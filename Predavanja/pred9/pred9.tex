\begin{frame}{State management: problem koji rješava}
Bez centralnog storea, dijeljenje podataka među komponentama zahtijeva \textbf{prop drilling}:

\vspace{0.4em}
\begin{center}
\texttt{App $\rightarrow$ Layout $\rightarrow$ Sidebar $\rightarrow$ UserInfo}
\end{center}

\vspace{0.4em}
Svaka komponenta u lancu mora primiti i proslijediti prop koji koristi samo zadnja. Ovo je:
\begin{itemize}
  \item Nepregledno - 5 razina prosljeđivanja za jedan podatak
  \item Krto - promjena na jednoj razini razbija lanac
  \item Teško za maintain - ne znaš odakle podatak dolazi
\end{itemize}

\vspace{0.6em}
Store rješava ovo: bilo koja komponenta može direktno čitati i pisati dijeljeni state bez posrednika.
\end{frame}

\begin{frame}{Optimistički vs pesimistički update}
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Pesimistički}
\begin{itemize}
  \item Pošalji request $\rightarrow$ čekaj odgovor $\rightarrow$ ažuriraj UI
  \item Korisnik vidi spinner
  \item Sigurniji - nema rollbacka
  \item Sporiji UX
\end{itemize}

\column{0.48\textwidth}
\textbf{Optimistički}
\begin{itemize}
  \item Ažuriraj UI odmah $\rightarrow$ pošalji request $\rightarrow$ rollback ako ne uspije
  \item Korisnik vidi instant promjenu
  \item Kompleksniji - treba rollback logika
  \item Brži UX
\end{itemize}
\end{columns}

\vspace{0.6em}
Za ovaj projekt koristimo pesimistički pristup - jednostavniji je i dovoljno brz. Optimistički update se koristi za akcije poput lajkanja objave ili micanja itema iz liste.
\end{frame}

// src/stores/lifters.ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import api from '@/api/client'
import type { Lifter } from '@/types'

export const useLifterStore = defineStore('lifters', () => {
  const lifters = ref<Lifter[]>([])
  const loading = ref(false)
  const error = ref<string | null>(null)

  async function fetchLifters(clubId: number) {
    loading.value = true
    error.value = null
    try {
      const { data } = await api.get<Lifter[]>(
        `/clubs/${clubId}/lifters`
      )
      lifters.value = data
    } catch {
      error.value = 'Greska pri dohvatu natjecatelja'
    } finally {
      loading.value = false
    }
  }

  return { lifters, loading, error, fetchLifters }
})
\end{lstlisting}
\end{frame}

\begin{frame}{Error stanja i retry}
Nije svaka greška ista. Razlikujemo:

\vspace{0.4em}
\begin{tabular}{@{}lp{4cm}p{5cm}@{}}
  \toprule
  \textbf{Tip} & \textbf{Primjer} & \textbf{Reakcija} \\
  \midrule
  Network error & Nema interneta & Retry gumb, provjeri konekciju \\
  Server error (5xx) & Bug na backendu & Retry gumb, \enquote{Pokušajte kasnije} \\
  Auth error (401) & Token istekao & Automatski refresh ili redirect \\
  Client error (4xx) & Krivi podaci & Prikaži poruku, ne nudi retry \\
  \bottomrule
\end{tabular}

\vspace{0.6em}
Retry ima smisla samo za tranzijentne greške (network, 5xx). Za 400/422/409 retry će dati isti rezultat - korisnik mora promijeniti input.
\end{frame}

% ------------------------------------------------------------
\begin{frame}{Debouncing: ne šalji request na svaki keystroke}
Kad korisnik piše u search polje, ne želimo slati API request na svako slovo:

\vspace{0.4em}
\begin{center}
\texttt{p $\rightarrow$ po $\rightarrow$ pow $\rightarrow$ powe $\rightarrow$ power} = 5 requestova za jednu riječ
\end{center}

\vspace{0.6em}
\textbf{Debounce} odgađa akciju dok korisnik ne prestane tipkati (npr. 300ms):
\begin{enumerate}
  \item Korisnik piše \texttt{p} $\rightarrow$ timer kreće (300ms)
  \item Piše \texttt{o} $\rightarrow$ timer se resetira
  \item Piše \texttt{w}, \texttt{e}, \texttt{r} $\rightarrow$ timer se resetira svaki put
  \item 300ms bez tipkanja $\rightarrow$ šalje se jedan request za \texttt{power}
\end{enumerate}

\vspace{0.4em}
Za frontend filtriranje (computed) debounce nije potreban jer nema network poziva. Za API search endpoint je obavezan.
\end{frame}

\begin{frame}[fragile,shrink=5]{Composable: useDebouncedSearch}
\begin{lstlisting}[language=typescript]
// src/composables/useDebouncedSearch.ts
import { ref, watch } from 'vue'

export function useDebouncedSearch(delay = 300) {
  const search = ref('')
  const debounced = ref('')
  let timer: ReturnType<typeof setTimeout>

  watch(search, (value) => {
    clearTimeout(timer)
    timer = setTimeout(() => { debounced.value = value }, delay)
  })

  return { search, debounced }
}

// u komponenti:
// const { search, debounced } = useDebouncedSearch()
// <input v-model="search" />  -> API poziv prati debounced
\end{lstlisting}

\vspace{0.4em}
Composable je funkcija s prefiksom \texttt{use} koja kapsulira reaktivnu logiku - isto što i Pinia store, ali bez globalnog stanja. Svaka komponenta koja je pozove dobiva svoju instancu. Ovo je centralni Composition API pattern za dijeljenje logike.
\end{frame}

import { ref } from 'vue'
import { useRouter } from 'vue-router'
import { AxiosError } from 'axios'
import api from '@/api/client'
import { useAuthStore } from '@/stores/auth'

const auth = useAuthStore()
const router = useRouter()
const form = ref({
  first_name: '', last_name: '',
  gender: 'M', birth_date: '',
})
const errors = ref<Record<string, string>>({})

async function submit() {
  errors.value = {}
  try {
    await api.post(
      `/clubs/${auth.user!.club_id}/lifters`, form.value
    )
    router.push('/lifters')
  } catch (e) {
    const err = e as AxiosError
    if (err.response?.status === 422) {
      errors.value = parseValidationErrors(err.response.data)
    }
  }
}
</script>

\begin{frame}{Validacija: klijentska vs serverska}
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Klijentska (frontend)}
\begin{itemize}
  \item HTML atributi: \texttt{required}, \texttt{min}, \texttt{maxlength}
  \item Brzi feedback - bez mrežnog poziva
  \item Može se zaobići (DevTools)
  \item Za UX, ne za sigurnost
\end{itemize}

\column{0.48\textwidth}
\textbf{Serverska (backend)}
\begin{itemize}
  \item Pydantic schema validacija
  \item Vraća 422 s detaljima
  \item Ne može se zaobići
  \item Autoritet za podatke
\end{itemize}
\end{columns}

\vspace{0.6em}
Trebate obje. Klijentska daje brzi feedback, serverska jamči integritet. Frontend prikazuje serverske greške kad klijentska validacija nije dovoljna (npr. duplikat u bazi).
\end{frame}

% ------------------------------------------------------------
\begin{frame}{Form UX best practices}
\begin{enumerate}
  \item \textbf{Inline validacija} - prikaži grešku ispod polja odmah kad korisnik napusti polje, ne tek na submit
  \item \textbf{Disable submit dok forma nije validna} - korisnik zna da nešto nedostaje
  \item \textbf{Clear error kad korisnik počne tipkati} - ne drži crveni tekst nakon što je korisnik počeo ispravljati
  \item \textbf{Focus na prvo polje s greškom} - nakon neuspjelog submita, premjesti fokus na problem
  \item \textbf{Serverske greške uz polje} - 422 greška za \texttt{birth\_date} ide pored datumskog inputa, ne kao generička poruka
\end{enumerate}
\end{frame}

\begin{frame}{Prikaz serverskih grešaka u formi}
Backend može vratiti dva tipa grešaka koje trebamo prikazati:

\vspace{0.4em}
\begin{tabular}{@{}llp{5.5cm}@{}}
  \toprule
  \textbf{Status} & \textbf{Tip} & \textbf{Prikaz} \\
  \midrule
  422 & Field-level validacija & Greška ispod odgovarajućeg polja (\texttt{loc} kaže koje) \\
  409 & Duplikat & Globalna poruka iznad forme: \enquote{Natjecatelj je već prijavljen} \\

\begin{frame}[fragile,shrink=5]{Parsiranje 422 grešaka s backenda}
\begin{lstlisting}[language=typescript]
// src/utils/errors.ts

// FastAPI 422 format:
// { "detail": [{ "loc": ["body","first_name"],
//   "msg": "field required", "type": "missing" }] }

interface ValidationDetail {
  loc: (string | number)[]
  msg: string
}

export function parseValidationErrors(
  data: { detail?: ValidationDetail[] } | unknown
): Record<string, string> {
  const errors: Record<string, string> = {}
  const detail = (data as { detail?: ValidationDetail[] })?.detail
  if (Array.isArray(detail)) {
    for (const err of detail) {
      // loc je ["body", "field_name"]
      const field = String(err.loc[err.loc.length - 1])
      errors[field] = err.msg
    }
  }
  return errors
}
\end{lstlisting}

\vspace{0.4em}
Svaka greška ima \texttt{loc} (lokacija polja) i \texttt{msg} (poruka). Mapiramo ih na objekt \texttt{\{field: message\}} za prikaz u formi.
\end{frame}

\begin{frame}{Kompozicija vs nasljeđivanje u UI-ju}
Vue ne koristi nasljeđivanje komponenti. Umjesto toga koristi \textbf{kompoziciju}:

\vspace{0.4em}
\begin{tabular}{@{}lp{5cm}p{5cm}@{}}
  \toprule
  & \textbf{Nasljeđivanje} & \textbf{Kompozicija} \\
  \midrule
  Pristup & \texttt{class AdminTable extends Table} & \texttt{<DataTable>} s propsima i slotovima \\
  Fleksibilnost & Rigidna hijerarhija & Slobodna kombinacija \\
  Reusability & Teško za različite kontekste & Lako - props i slots \\
  \bottomrule
\end{tabular}

\vspace{0.6em}
Slots omogućuju parentu da definira \textbf{što} se prikazuje, a child komponenta definira \textbf{gdje} i \textbf{kako}. Ovo je isti princip kao \enquote{composition over inheritance} iz OOP-a.
\end{frame}

\begin{frame}{Props dizajn: API vaše komponente}
Propsi su \textbf{javni API} vaše komponente. Dizajnirajte ih s istom pažnjom kao REST endpoint:

\vspace{0.4em}
\begin{itemize}
  \item \textbf{Imenujte opisno}: \texttt{isLoading} umjesto \texttt{flag}, \texttt{errorMessage} umjesto \texttt{msg}
  \item \textbf{Definirajte tipove}: generički \texttt{defineProps<\{...\}>()} - TypeScript provjerava svaki prop na mjestu korištenja
  \item \textbf{Postavite defaulte}: \texttt{withDefaults()} za opcionalne propse
  \item \textbf{Koristite v-model za dvosmjerne propse}: umjesto prop + emit pattern ručno
\end{itemize}

\vspace{0.6em}
Komponenta s 10+ propsova je znak da radi previše. Razmislite o razbijanju na manje komponente ili korištenju slotova.
\end{frame}

\begin{frame}{Key prop u v-for: zašto je obavezan}
Bez \texttt{:key}, Vue koristi \enquote{in-place patch} strategiju - ažurira postojeće DOM elemente umjesto da ih premjesti.

\vspace{0.4em}
Problem: korisnik uređuje input u 2. redu tablice. Filtriranjem se 2. red promijeni - ali DOM element ostane isti. Input vrijednost pripada starom redu.

\vspace{0.6em}
\begin{tabular}{@{}lp{8cm}@{}}
  \toprule
  \textbf{Key} & \textbf{Ponašanje} \\
  \midrule
  \texttt{:key="item.id"} & Vue prati identitet elementa - premješta DOM node kad se lista presloži \\
  \texttt{:key="index"} & Vue tretira poziciju kao identitet - bugovi pri sortiranju/filtriranju \\
  Bez keya & Warning u konzoli, nepredvidljivo ponašanje \\
  \bottomrule
\end{tabular}

\vspace{0.4em}
Koristite jedinstven, stabilan identifikator (ID iz baze) kao key. Nikad index.
\end{frame}

\begin{frame}{Emit pattern: inverzija kontrole}
Child komponenta ne odlučuje što se događa kad korisnik klikne gumb - samo javlja parentu.

\vspace{0.6em}
\begin{center}
\texttt{Child: emit('delete', itemId)} $\rightarrow$ \texttt{Parent: handleDelete(itemId)}
\end{center}

\vspace{0.6em}
\begin{itemize}
  \item Child ne zna koristi li se u admin viewu ili club viewu
  \item Parent odlučuje: prikaži modal? Odmah obriši? Zatraži razlog?
  \item Ista child komponenta se koristi u različitim kontekstima s različitim ponašanjem
\end{itemize}

\vspace{0.6em}
Ovo je \textbf{inverzija kontrole} - isti princip kao dependency injection na backendu. Komponenta prima ponašanje izvana umjesto da ga hardkodira.
\end{frame}

\begin{frame}[fragile,shrink=5]{Reusable FormField komponenta}
\begin{lstlisting}[language=html]
<!-- src/components/FormField.vue -->
<script setup lang="ts">
defineProps<{
  label: string
  error?: string
  modelValue: string | number
}>()
defineEmits<{
  'update:modelValue': [value: string]
}>()
</script>

<template>
  <div class="form-field">
    <label>{{ label }}</label>
    <input :value="modelValue"
      @input="$emit('update:modelValue',
        ($event.target as HTMLInputElement).value)"
    />
    <span v-if="error" class="error">{{ error }}</span>
  </div>
</template>
